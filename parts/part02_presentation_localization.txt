    local cache,count={},0
    local prefixes,suffixes={},{}
    local function exact(s) local p=dictionary[s]; return p and p[api.lang] end
    local function indexCatalog()
        prefixes={}; suffixes={}
        for source,pair in pairs(dictionary) do
            if #source>=4 and pair[api.lang]~=source then
                if source:match("[%s:=]$") and not source:find("%%") then
                    local first=source:sub(1,1); prefixes[first]=prefixes[first] or {}; table.insert(prefixes[first],source)
                end
                if source:match("^%s") and not source:find("%%") then suffixes[#suffixes+1]=source end
            end
        end
        for _,list in pairs(prefixes) do table.sort(list,function(a,b) return #a>#b end) end
        table.sort(suffixes,function(a,b) return #a>#b end)
    end
    local function translate(s,depth)
        if type(s)~="string" or s=="" then return s end
        local direct=exact(s); if direct then return direct end
        if depth>5 then return s end
        local visible,id=s:match("^(.-)(##.*)$")
        if visible then return translate(visible,depth+1)..id end
        local decoration,content=s:match("^([>x]%s+)(.+)$")
        if decoration then return decoration..translate(content,depth+1) end
        local leading,body,trailing=s:match("^(%s*)(.-)(%s*)$")
        if body~="" and body~=s then return leading..translate(body,depth+1)..trailing end
        -- Compose UI paths; names and numbers that are not catalog entries stay intact.
        if s:find(" / ",1,true) then
            local out={}; local from=1
            while true do
                local a,b=s:find(" / ",from,true)
                if not a then out[#out+1]=translate(s:sub(from),depth+1); break end
                out[#out+1]=translate(s:sub(from,a-1),depth+1); out[#out+1]=" / "; from=b+1
            end
            return table.concat(out)
        end
        local head=s:match("^(.-) %[%s*indisponivel%s*%]$")
        if head then return translate(head,depth+1)..(api.lang=="en-US" and " [unavailable]" or " [indisponível]") end
        for _,prefix in ipairs(prefixes[s:sub(1,1)] or {}) do
            if s:sub(1,#prefix)==prefix then return exact(prefix)..translate(s:sub(#prefix+1),depth+1) end
        end
        for _,suffix in ipairs(suffixes) do
            if s:sub(-#suffix)==suffix then return translate(s:sub(1,#s-#suffix),depth+1)..exact(suffix) end
        end
        return s
    end
    function api.text(s)
        if type(s)~="string" then return s end
        if cache[s] then return cache[s] end
        local value=translate(s,0)
        if count>=4096 then cache={}; count=0 end
        cache[s]=value; count=count+1
        return value
    end
    function api.list(list)
        local out={}
        for k,v in pairs(list) do
            -- Rosters contain player data, not UI vocabulary.
            out[k]=type(v)=="string" and not v:match("^%d+%s*[-:]") and api.text(v) or v
        end
        return out
    end

        local ok,v=pcall(FileMgr.ReadFileContent,readPath)
        if ok and type(v)=="string" and #v<64 then
            local lang=v:match("^language=([%a%-]+)%s*$")
            if lang=="en-US" or lang=="pt-BR" then api.lang=lang end
        end
    end
    indexCatalog()
    function api.change(lang)
        if lang~="pt-BR" and lang~="en-US" then return false end
        api.lang=lang; cache={}; count=0; indexCatalog()
        for _,entry in pairs(api.owned) do
            if entry.object.SetName then entry.object:SetName(api.text(entry.name)) end
            if entry.object.SetDesc then entry.object:SetDesc(api.text(entry.desc)) end
            if entry.options then
                local index=entry.object.GetListIndex and entry.object:GetListIndex()
                entry.object:SetList(api.list(entry.options))
                if index~=nil and entry.object.SetListIndex then entry.object:SetListIndex(index) end
            end
        end
        if prefPath and FileMgr.WriteFileContent then
            local ok,result=pcall(FileMgr.WriteFileContent,prefPath,"language="..lang.."\n",false)
            if ok and result==true then meloDropLegacy("MeloCatScrLanguage.ini") end
            return ok and result==true
        end
        return false
    end
    api.FeatureMgr=setmetatable({}, {__index=baseFeature})
    api.FeatureMgr.AddFeature=function(id,name,kind,desc,callback,...)
        local object=baseFeature.AddFeature(id,api.text(name),kind,api.text(desc),callback,...)
        local entry={object=object,name=name,desc=desc or ""}; api.owned[id]=entry
        local proxy={}
        setmetatable(proxy,{__index=function(_,key)
            local value=object[key]
            if type(value)~="function" then return value end
            if key=="SetName" or key=="SetDesc" then return function(_,v)
                entry[key=="SetName" and "name" or "desc"]=v
                value(object,api.text(v)); return proxy
            end end
            if key=="SetList" then return function(_,v)
                entry.options={}; for i,s in pairs(v) do entry.options[i]=s end
                value(object,api.list(v)); return proxy
            end end
            return function(_,...)
                local result=table.pack(value(object,...))
                if result[1]==object then result[1]=proxy end
                return table.unpack(result,1,result.n)
            end
        end})
        return proxy
    end
    api.FeatureMgr.RemoveFeature=function(id,...)
        api.owned[id]=nil; return baseFeature.RemoveFeature(id,...)
    end
    api.ImGui=setmetatable({}, {__index=ImGui})
    for method,index in pairs({Text=1,TextDisabled=1,TextWrapped=1,TextColored=5,Begin=1,BeginTabItem=1,CollapsingHeader=1,InputText=1,AddText=3}) do
        local original=ImGui[method]
        if original then api.ImGui[method]=function(...)
            local args=table.pack(...); args[index]=api.text(args[index]); return original(table.unpack(args,1,args.n))
        end end
    end
    api.GUI=setmetatable({}, {__index=baseGUI})
    api.GUI.AddToast=function(title,text,...) return baseGUI.AddToast(api.text(title),api.text(text),...) end
    api.string=setmetatable({}, {__index=baseString})
    api.string.format=function(fmt,...) return baseString.format(api.text(fmt),...) end
    -- Cherax owns native feature labels, lists and tooltip storage. Never mutate them during rendering.
    -- Lua-owned features are translated once when created or when the selected language changes.
    api.ClickGUI=setmetatable({}, {__index=baseClick})
    api.ClickGUI.RenderFeature=function(id,...)
        return baseClick.RenderFeature(id,...)
    end
    return api
end)()
local FeatureMgr,ImGui,GUI,ClickGUI,string=I18N.FeatureMgr,I18N.ImGui,I18N.GUI,I18N.ClickGUI,I18N.string
-- LANGUAGE END
local gunsmithScreenActive=false

    -- Public bindings -------------------------------------------------------
    return {
        ClickGUI = { get = function() return ClickGUI end, set = function(value) ClickGUI = value end },
        FeatureMgr = { get = function() return FeatureMgr end, set = function(value) FeatureMgr = value end },
        GUI = { get = function() return GUI end, set = function(value) GUI = value end },
        I18N = { get = function() return I18N end, set = function(value) I18N = value end },
        ImGui = { get = function() return ImGui end, set = function(value) ImGui = value end },
        gunsmithScreenActive = { get = function() return gunsmithScreenActive end, set = function(value) gunsmithScreenActive = value end },
        string = { get = function() return string end, set = function(value) string = value end }
    }
end
