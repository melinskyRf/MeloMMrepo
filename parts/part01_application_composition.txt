factories["application/diagnostics"]=(function()
return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV = context:scope("application/diagnostics", {
        ["GUI"] = "presentation/localization",
        ["I18N"] = "presentation/localization",
        ["util"] = "application/task_state"
    })


    -- Implementation --------------------------------------------------------
local function now_ms() return Time.GetEpocheMs() end
-- BRAND LABEL BEGIN
local function brandLabel(text)
    return tostring(text or ""):gsub("MeloCatScr Script","MeloCatScr")
end
-- BRAND LABEL END
local function toast(text) GUI.AddToast("MeloCatScr", brandLabel(text), 5000) end
util.toast = toast
util.joaat = Utils.Joaat
local errorHistory={}
local function report(text)
    local full=tostring(text)
    if Logger and Logger.LogInfo then pcall(Logger.LogInfo,"[MeloCatScr] "..brandLabel(full)) end
    errorHistory[#errorHistory+1]=full:sub(1,16000)
    while #errorHistory>20 do table.remove(errorHistory,1) end
    if FileMgr and FileMgr.WriteFileContent then
        local errPath=meloDataPath("MeloCatScrErrors.txt")
        if errPath then
            local ok,result=pcall(FileMgr.WriteFileContent,errPath,"MeloCatScr Revision=2026-09-29.4\n\n"..table.concat(errorHistory,"\n\n"),false)
            if ok and result==true then meloDropLegacy("MeloCatScrErrors.txt") end
        end
    end
    local first=full:match("[^\r\n]+") or full
    toast(first:sub(1,180).."\n"..I18N.text("Details saved to MeloCatScrErrors.txt"))
end

    -- Public bindings -------------------------------------------------------
    return {
        brandLabel = { get = function() return brandLabel end, set = function(value) brandLabel = value end },
        now_ms = { get = function() return now_ms end, set = function(value) now_ms = value end },
        report = { get = function() return report end, set = function(value) report = value end },

    })


    -- Implementation --------------------------------------------------------
local function enqueue(fn, ref)
    local job={thread=coroutine.create(fn),due=0,ref=ref,generation=ref and ref.generation}
    jobs[#jobs+1]=job
    return job
end
function util.yield(ms)
    if runningJob and coroutine.running()==runningJob.thread then return coroutine.yield(ms or 0) end
    return Script.Yield(ms or 0)
end
util.yield_once = util.yield
function util.stop_thread() coroutine.yield("stop") end
function util.create_tick_handler(fn) tickers[#tickers + 1] = {fn = fn} end
function util.on_stop(fn) cleanups[#cleanups + 1] = fn end

    -- Public bindings -------------------------------------------------------
    return {
        enqueue = { get = function() return enqueue end, set = function(value) enqueue = value end }
    }
end

-- ----------------------------------------------------------------------------
-- [MC:application/task_engine] 01.06 / TASK ENGINE
-- Advance scheduled tasks and handle cancellation and recovery.
-- ----------------------------------------------------------------------------
factories["application/task_engine"]=(function()
-- One execution engine owns scheduling, cancellation, wakeups and callback failure.
return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV=context:scope("application/task_engine", {
        ["deleteOwned"]="world/entity_registry",
        ["enqueue"]="application/task_dispatcher",
        ["jobs"]="application/task_state",
        ["players"]="world/roster_state",
        ["releaseJob"]="application/task_release",
        ["report"]="application/diagnostics",
        ["runningJob"]="application/task_state",
        ["stopped"]="application/task_state",
        ["tickers"]="application/task_state",
    })

    -- Implementation --------------------------------------------------------
    local TaskEngine={}
    local function current(job)
        local owner=job.ref
        return not owner or (job.generation==owner.generation and (not owner.pid or players.exists(owner.pid)))
    end
    local function recover(job,message)
        local owner=job.ref
        report((owner and owner.path..": " or "")..tostring(message))
        if job.ticker then job.ticker.disabled=true end
        if not owner then return end
        owner.generation=owner.generation+1
        local toggle=owner.kind==eFeatureType.Toggle
        if toggle then owner.value=false; owner.feature:SetBoolValue(false) end
        enqueue(function()
            deleteOwned(owner)
            if toggle then owner.callback(false) end
        end)
    end
    local function advance(job,time)
        if not current(job) then releaseJob(job); return end
        if time<job.due then jobs[#jobs+1]=job; return end
        runningJob=job
        local ok,result=coroutine.resume(job.thread)
        runningJob=nil
        if not ok then recover(job,result) end
        if not ok or result=="stop" or coroutine.status(job.thread)=="dead" then
            releaseJob(job)
        else
            job.due=time+math.max(0,tonumber(result) or 0)
            jobs[#jobs+1]=job
        end
    end
    function TaskEngine.pump(time)
        if stopped then return false end
        for _,ticker in ipairs(tickers) do
            if not ticker.busy and not ticker.disabled then
                ticker.busy=true
                local job=enqueue(function() if ticker.fn()==false then ticker.disabled=true end end)
                job.ticker=ticker
            end
        end
        local batch=jobs
        jobs={}
        for _,job in ipairs(batch) do advance(job,time) end
        return true
    end
    function TaskEngine.cancelPending()
        stopped=true
        local pending=jobs; jobs={}
        for _,job in ipairs(pending) do releaseJob(job) end
    end
    -- Public bindings -------------------------------------------------------
    return {taskEngine={get=function() return TaskEngine end}}
end

end)()

return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV = context:scope("application/task_release", {
        ["allocations"] = "world/resource_state",
        ["memory"] = "world/resource_state"
    })


    -- Implementation --------------------------------------------------------
local function releaseJob(job)
    if job.finalize then pcall(job.finalize); job.finalize=nil end
    for ptr, owner in pairs(allocations) do if owner == job then memory.free(ptr) end end
    if job.ticker then job.ticker.busy = false end
    if coroutine.status(job.thread) ~= "dead" then coroutine.close(job.thread) end
end

    -- Public bindings -------------------------------------------------------
    return {
        releaseJob = { get = function() return releaseJob end, set = function(value) releaseJob = value end }
    }
end

-- Descriptor-based control creation, with one event dispatcher for all controls.
return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV=context:scope("presentation/feature_runtime", {
        ["FeatureMgr"]="presentation/localization",
        ["aliases"]="presentation/workspace_state",
        ["brandLabel"]="application/diagnostics",
        ["deleteOwned"]="world/entity_registry",
        ["enqueue"]="application/task_dispatcher",
        ["layoutRevision"]="presentation/workspace_state",
        ["players"]="world/roster_state",
        ["refs"]="presentation/workspace_state",
        ["stopped"]="application/task_state",
    })

    -- Implementation --------------------------------------------------------
    local function scheduleTransition(control,value)
        if control.value==value then return false end
        control.value=value
        control.generation=control.generation+1
        enqueue(function()
            deleteOwned(control)
            control.callback(value)
        end,control)
        return true
    end
    local function dispatch(control,feature)
        if stopped or (control.pid and not players.exists(control.pid)) then return end
        local state=context:read("presentation/workspace_state","panel")
        for i=#state.recent,1,-1 do if state.recent[i]==control.id then table.remove(state.recent,i) end end
        table.insert(state.recent,1,control.id);if #state.recent>12 then table.remove(state.recent) end
        if control.kind==eFeatureType.Toggle then
            scheduleTransition(control,feature:IsToggled())
        elseif control.callback then
            enqueue(function() control.callback(feature) end,control)
        end
    end
    local function registerControl(spec)
        assert(spec.parent and spec.parent.items,"Control needs a registration owner")
        local path=spec.parent.path.."/"..spec.name
        local id=Utils.Joaat(path)
        assert(not refs[id],"Duplicate control ID: "..path)
        local control={id=id,name=brandLabel(spec.name),path=path,parent=spec.parent,pid=spec.parent.pid,
            generation=0,value=false,kind=spec.kind,callback=spec.execute}
        local feature=FeatureMgr.AddFeature(id,control.name,spec.kind,brandLabel(spec.description),
            function(f) dispatch(control,f) end,true)
        control.feature=feature
        feature:SetSaveable(false)
        if spec.kind==eFeatureType.Toggle then feature:SetBoolValue(false) end
        refs[id]=control
        spec.parent.items[#spec.parent.items+1]=control
        for _,alias in ipairs(spec.commands or {}) do if alias~="" then aliases[alias]=control end end
        layoutRevision=layoutRevision+1
        return control
    end
    -- Public bindings -------------------------------------------------------
    return {registerControl={get=function() return registerControl end},scheduleTransition={get=function() return scheduleTransition end}}
end

end)()

local function addAvailabilityReview(misc)
    local page=menu.list(misc,"Compatibilidade",{},"Revisa controles e grava os motivos das limitacoes.")
    misc.viewTabs[#misc.viewTabs+1]={name="Compatibilidade",node=page}
    local candidates={}
    local specs={
        ["Custom Notification"]={"Notification","Send Notification","Enviar aviso"},
        ["Custom Label"]={"Notify Label","Send Notify Label","Enviar rotulo"},
        ["SMS Spam"]={"Text Message","Send Text Message","SMS"},
    }
    local function discover()
        for _,r in pairs(refs) do if r.unavailableReason and not r.supportRefresh and not candidates[r.id] then
            local old=r.path:match("/([^/]+) %[%s*indisponivel%s*%]$")
            local c={ref=r,spec=specs[old],reason=r.unavailableReason}; candidates[r.id]=c
            if c.spec then
                c.input=create(r.parent,c.spec[3].." / texto",eFeatureType.InputText,"Texto do aviso ou codigo de rotulo GTA. Envio manual ao jogador selecionado.")
                c.input.feature:SetStringValue("")
                local group={name=c.spec[3]}; c.input.configGroup=group; r.configGroup=group
            end
        end end
    end
    local function exact(pid,name,kind)
        return resolveNative(name,kind,pid)
    end
    local function refreshOne(c)
        local r,s=c.ref,c.spec
        if not s then return end
        local input=exact(r.pid,s[1],eFeatureType.InputText)
        local send=exact(r.pid,s[2],eFeatureType.Button)
        local isLocal=r.pid==players.user() and s[1]~="Text Message"
        c.available=isLocal or input~=nil and send~=nil
        local label=s[3]..(c.available and "" or " [indisponivel]")
        if r.name~=label then displayName(r,label) end
        r.unavailableReason=c.available and nil or ("Cherax precisa expor "..s[1].." (texto) e "..s[2].." (botao) para este jogador.")
        if r.feature.SetDesc then r.feature:SetDesc(r.unavailableReason or (isLocal and "Shows a notification on your own screen." or "Usa os controles atuais do Cherax. O destinatario pode bloquear a mensagem.")) end
        -- Resolve again at execution; never hold stale native feature objects across sessions.
        r.recovered=function()
            refreshOne(c)
            if not c.available then toast(r.unavailableReason); return end
            if not isLocal and (not players.exists(r.pid) or not NETWORK.NETWORK_IS_IN_SESSION() or NETWORK.NETWORK_IS_IN_TRANSITION()
                or NETWORK.NETWORK_IS_SESSION_BUSY()) then toast(I18N.text("Aguarde uma sessao estavel.")); return end
            if c.next and now_ms()<c.next then toast(I18N.text("Aguarde tres segundos entre avisos.")); return end
            local value=c.input.feature:GetStringValue():gsub("[%c]"," ")
            if not value:find("%S") or #value>140 then toast(I18N.text("Use de 1 a 140 bytes de texto.")); return end
            if s[1]=="Notify Label" and not value:match("^[%w_]+$") then toast(I18N.text("Use um codigo de rotulo GTA, como uma chave de texto; sem espacos.")); return end
            if isLocal then
                if s[1]=="Notify Label" then
                    value=HUD.GET_FILENAME_FOR_AUDIO_CONVERSATION(value)
                    if not value or value=="NULL" or value=="" then toast("Unknown GTA text label."); return end
                end
                c.next=now_ms()+3000
                HUD.BEGIN_TEXT_COMMAND_THEFEED_POST("STRING")
                -- Split only at UTF-8 boundaries; the GTA text component has a byte limit.
                while #value>0 do
                    local last=math.min(90,#value)
                    while last<#value and value:byte(last+1)>=128 and value:byte(last+1)<192 do last=last-1 end
                    HUD.ADD_TEXT_COMPONENT_SUBSTRING_PLAYER_NAME(value:sub(1,last)); value=value:sub(last+1)
                end
                HUD.END_TEXT_COMMAND_THEFEED_POST_TICKER(false,false)
                return
            end
            local f,b=exact(r.pid,s[1],eFeatureType.InputText),exact(r.pid,s[2],eFeatureType.Button)
            if not f or not b then toast(I18N.text("O controle do Cherax mudou. Atualize a compatibilidade.")); return end
            local old=f:GetStringValue()
            c.next=now_ms()+3000
            local ok=pcall(function() f:SetStringValue(value); b:TriggerCallback() end)
            pcall(function() f:SetStringValue(old) end)
            toast(ok and "Envio solicitado ao Cherax. O recebimento depende do destinatario." or "Cherax nao conseguiu solicitar o envio.")
        end
    end
    local lastWritten
    local function review(write,quiet)
        discover()
        local lines={"MeloCatScr - compatibilidade | Revision=2026-09-29.4","Controles da API sao conferidos por nome, tipo e jogador."}
        local restored,blocked=0,0
        for _,c in pairs(candidates) do
            if c.spec then refreshOne(c) end
            local reason=c.ref.unavailableReason or c.reason
            if c.available then restored=restored+1 else blocked=blocked+1 end
            lines[#lines+1]=(c.available and "DISPONIVEL | " or "SEM SUPORTE | ")..c.ref.path.." | "..(c.available and "Controle original do Cherax." or reason)
        end
        for _,r in pairs(refs) do if r.supportRefresh then
            local f=r.supportRefresh()
            lines[#lines+1]=(f and "DISPONIVEL | " or "SEM SUPORTE | ")..r.path.." | "..(r.supportSource or "Controle atual do Cherax; requisitos de sessao continuam aplicaveis.")
        end end
        table.sort(lines, function(a,b) return a<b end)
        if write then
            local path=meloDataPath("MeloCatScrCompatibilidade.txt")
            local ok,result=false,false
            local content=table.concat(lines,"\n")
            if content==lastWritten then ok,result=true,true
            elseif path and FileMgr.WriteFileContent then
                ok,result=pcall(FileMgr.WriteFileContent,path,content,false)
                if ok and result==true then lastWritten=content;meloDropLegacy("MeloCatScrCompatibilidade.txt") end
            end
            if quiet then return end
            toast(string.format("Revisao: %d avisos integrados; %d controles antigos sem suporte. %s",restored,blocked,ok and result==true and "Relatorio salvo em Lua/MeloCatScr/MeloCatScrCompatibilidade.txt." or "Nao foi possivel salvar o relatorio."))
        end
    end
    menu.action(page,"Revisar suporte",{},"Confere todas as opcoes antigas e os destinos Cherax. Salva os motivos em MeloCatScrCompatibilidade.txt.",function() review(true) end)
    menu.action(page,"Ver limitacoes",{},"Explica as opcoes que continuam bloqueadas.",function()
        toast(I18N.text("Eventos antigos de propriedades, spam e remocao de godmode continuam bloqueados. Transaction Error dependia de atividade removida. Nasa Kick e Quandale Dingle nao possuem substituto validado."))
    end)
    review(true,true)
    local nextReport=now_ms()+10000
    local lastPid,nextCheck=nil,0
    util.create_tick_handler(function()
        if now_ms()<nextCheck then return end; nextCheck=now_ms()+2000
        discover()
        if now_ms()>=nextReport then nextReport=now_ms()+10000; review(true,true) end
        local pid=panel.selectedPlayer
        if pid~=nil then
            lastPid=pid
            for _,c in pairs(candidates) do if c.ref.pid==pid then refreshOne(c) end end
            for _,r in pairs(refs) do if r.pid==pid and r.supportRefresh then r.supportRefresh() end end
        end
    end)
end

local function addPresentation(misc)
    local prefs={notices=1,tips=1,seconds=6,side=1}
    local path=meloDataPath("MeloCatScrVisual.ini")
    local readPath=meloReadPath("MeloCatScrVisual.ini")
    if readPath and FileMgr.ReadFileContent and (not FileMgr.DoesFileExist or FileMgr.DoesFileExist(readPath)) then
        local ok,value=pcall(FileMgr.ReadFileContent,readPath)
        if ok and type(value)=="string" and #value<256 then
            for key,n in value:gmatch("(%a+)=(%d+)") do
                n=tonumber(n)
                if prefs[key]~=nil and ((key=="seconds" and n>=3 and n<=12) or (key~="seconds" and n<=1)) then prefs[key]=n end
            end
        end
    end
    local originalToast=toast
    local function save()
        if not path or not FileMgr.WriteFileContent then originalToast("Preferencias visuais: gravacao indisponivel."); return end
        local ok,v=pcall(FileMgr.WriteFileContent,path,string.format("notices=%d\ntips=%d\nseconds=%d\nside=%d\n",prefs.notices,prefs.tips,prefs.seconds,prefs.side),false)
        if ok and v==true then meloDropLegacy("MeloCatScrVisual.ini")
        else originalToast("Nao foi possivel salvar as preferencias visuais.") end
    end
    local page=menu.list(misc,"Notificacoes",{},"Avisos locais na lateral, sem enviar mensagens para outros jogadores.")
    misc.viewTabs[#misc.viewTabs+1]={name="Notificacoes",node=page}
    local pending,active={},{}
    local failed=false
    local function supported() return ImGui.AddText and ImGui.AddRectFilled and ImGui.GetDisplaySize end
    local function push(value)
        if stopped or prefs.notices==0 or failed or not supported() then return false end
        local text=brandLabel(value):gsub("[%c]"," ")
        for _,q in ipairs(active) do if q.text==text then return true end end
        for _,q in ipairs(pending) do if q.text==text then return true end end
        if #pending>=8 then originalToast(value); return true end
        pending[#pending+1]={text=text}
        return true
    end
    toast=function(value) if not push(value) then originalToast(value) end end
    util.toast=toast
    local function toggle(parent,label,key,help)
        local r=menu.toggle(parent,label,{},help,function(on)
            prefs[key]=on and 1 or 0
            if key=="notices" and not on then pending={}; active={} end
            save()
        end)
        r.value=prefs[key]==1; r.feature:SetBoolValue(r.value)
    end
    toggle(page,"Avisos laterais","notices","Mostra as mensagens do script em ate tres cartoes. Desligado: usa os avisos padrao do Cherax.")
    toggle(page,"Dicas iniciais","tips","Mostra duas dicas ao iniciar; nao repete durante a sessao.")
    local duration=create(page,"Duracao",eFeatureType.SliderInt,"Segundos por aviso.",function(f) prefs.seconds=math.max(3,math.min(12,f:GetIntValue())); save() end)
    duration.feature:SetLimitValues(3,12):SetStepSize(1):SetIntValue(prefs.seconds)
    local side=create(page,"Lado",eFeatureType.Combo,"Lateral da tela para os avisos.",function(f) prefs.side=math.max(0,math.min(1,f:GetListIndex())); save() end)
    side.feature:SetList({"Esquerda","Direita"}):SetListIndex(prefs.side)
    menu.action(page,"Testar aviso",{},"Mostra um cartao de exemplo.",function() toast(I18N.text("Visual pronto. Ajuste o lado e a duracao em Configuracoes > Notificacoes.")) end)
    menu.action(page,"Limpar avisos",{},"Fecha os avisos e limpa a fila.",function() pending={}; active={} end)
    -- UTF-8 characters stay intact, including long names. The full message is logged.
    local function wrap(text,limit)
        local chars={}
        local ok=pcall(function() for _,cp in utf8.codes(text) do chars[#chars+1]=utf8.char(cp) end end)
        if not ok then chars={}; for c in text:gmatch(".") do chars[#chars+1]=c:match("[%w%p%s]") and c or "?" end end
        local lines={}; local n=math.min(#chars,limit*4)
        for start=1,n,limit do lines[#lines+1]=table.concat(chars,"",start,math.min(n,start+limit-1)) end
        if #chars>n then lines[#lines]=lines[#lines]:sub(1,-1).."..." end
        return lines
    end
    local launched=now_ms()
    local tips={
        {at=2000,text="Bem-vindo ao MeloCatScr. Acoes ficam a esquerda; ajustes ficam a direita."},
        {at=9000,text="Seu visual: logo fixa no topo. Efeitos e avisos podem ser ajustados em Configuracoes."},
    }
    local function draw()
        if gunsmithScreenActive or stopped or prefs.notices==0 or failed or not supported() then return end
        local w,h=ImGui.GetDisplaySize(); if not w or not h or w<320 or h<200 then return end
        local now=now_ms()
        for _,tip in ipairs(tips) do if not tip.done and now-launched>=tip.at then tip.done=true; if prefs.tips==1 then push(tip.text) end end end
        for i=#active,1,-1 do if now>=active[i].ends then table.remove(active,i) end end
        local capacity=math.max(1,math.min(3,math.floor((h-90)/120)))
        while #active<capacity and #pending>0 do
            local q=table.remove(pending,1); q.start=now; q.ends=now+prefs.seconds*1000
            active[#active+1]=q
            if Logger and Logger.LogInfo then Logger.LogInfo("[MeloCatScr aviso] "..q.text) end
        end
        local width=math.min(340,w-32)
        local limit=math.max(20,math.floor((width-32)/8))
        local y=math.max(24,h*0.12)
        local acc=uiAccent()
        local ar,ag,ab=math.floor(acc[1]*255+0.5),math.floor(acc[2]*255+0.5),math.floor(acc[3]*255+0.5)
        for _,q in ipairs(active) do
            local t=math.min(1,(now-q.start)/300,(q.ends-now)/400)
            local alpha=math.floor(math.max(0,t)*255)
            local slide=(1-math.max(0,t))*(width+20)
            local x=prefs.side==1 and w-width-16+slide or 16-slide
            local lines=wrap(I18N.text(q.text),limit); local height=45+#lines*16
            -- Cartao de aviso: fundo painel; barra lateral, titulo e barra de tempo na cor de destaque atual.
            ImGui.AddRectFilled(x,y,x+width,y+height,18,19,26,math.floor(alpha*0.96),8)
            ImGui.AddRectFilled(x,y+8,x+3,y+height-8,ar,ag,ab,alpha,1)
            ImGui.AddText(x+14,y+9,"MeloCatScr",ar,ag,ab,alpha)
            for i,line in ipairs(lines) do ImGui.AddText(x+14,y+15+i*16,line,230,230,230,alpha) end
            local remain=math.max(0,(q.ends-now)/(q.ends-q.start))
            ImGui.AddRectFilled(x+14,y+height-7,x+14+(width-28)*remain,y+height-5,ar,ag,ab,alpha,1)
            y=y+height+10
        end
    end
    local id=EventMgr.RegisterHandler(eLuaEvent.ON_PRESENT,function()
        local ok,err=pcall(draw)
        if not ok and not failed then
            failed=true
            for _,q in ipairs(active) do originalToast(q.text) end
            for _,q in ipairs(pending) do originalToast(q.text) end
            active={}; pending={}
            originalToast("Avisos laterais desativados nesta execucao: "..tostring(err))
        end
    end)
    eventIds[#eventIds+1]=id
    util.on_stop(function() pending={}; active={}; toast=originalToast; util.toast=originalToast end)

end

        end
    end

        if not file or not FileMgr.WriteFileContent then toast(I18N.text("Logo: gravacao de preferencias indisponivel.")); return end

    local textureId,requested,failed,start,preview=nil,false,false,nil,false
    local deadline=0
    local function loadTexture()

        requested=true; deadline=now_ms()+10000
        if not Texture or not Texture.LoadTextureAsync or not Texture.IsTextureValid or not Texture.GetTexture

    -- Animated frame-sequence loader (frames 015..050) ------------------------
    local LOADER_FPS=12 -- velocidade da animacao dos frames; teste 8, 10, 12, 24, 30, 60
    local LOADER_TAIL=10 -- ultimos frames da sequencia rodam mais devagar (aterrissagem suave)
    local LOADER_TAIL_DIV=2 -- divide o fps nesses frames (2 = metade da velocidade)
    local FRAME_FIRST,FRAME_LAST=15,50
    local FRAME_TOTAL=FRAME_LAST-FRAME_FIRST+1
    local LOADER_FRAMES_DIR="" -- vazio = deteccao automatica junto do script; ou caminho da pasta dos frames (ou da pasta pai)
    local frames,frameAspect={},nil
    local framesState="idle" -- idle|loading|ready|failed
    local framesMissing,framesDeadline=0,0
    local frameIndex,frameAccum,frameTick=1,0,nil
    -- Area de conteudo visivel: uniao das bounding boxes dos 36 frames (medida) + folga de 6px.
    -- Recorta as margens transparentes para a logo ocupar o quadro como a logo estatica.
    local FRAME_UV={0.2828,0.0139,0.7203,0.875}
    local PROBE_CHECKS={
        {"\\PHOTS\\ezgif-frame-%03d.png","\\PHOTS","ezgif-frame-%03d.png"},
        {"\\PHOTS\\ezgif-frame-%03d.jpg","\\PHOTS","ezgif-frame-%03d.jpg"},
        {"\\ezgif-frame-%03d.png","","ezgif-frame-%03d.png"},
        {"\\frames\\ezgif-frame-%03d.png","\\frames","ezgif-frame-%03d.png"},
        {"\\frames_sem_fundo\\ezgif-frame-%03d.png","\\frames_sem_fundo","ezgif-frame-%03d.png"},
        {"\\ezgif-frame-%03d.jpg","","ezgif-frame-%03d.jpg"},
        {"\\frames\\ezgif-frame-%03d.jpg","\\frames","ezgif-frame-%03d.jpg"},
        {"\\%03d.png","","%03d.png"},
        {"\\frames\\%03d.png","\\frames","%03d.png"},
        {"\\%03d.jpg","","%03d.jpg"},
    }
    local function logLoader(msg)
        if Logger and Logger.LogInfo then pcall(Logger.LogInfo,"[Loader] "..msg) end
    end
    local function resetFrameClock()
        frameIndex=1; frameAccum=0; frameTick=nil
    end
    local function scriptDir()
        if not (debug and debug.getinfo) then return nil end
        local ok,src=pcall(function() return debug.getinfo(1,"S").source end)
        if not ok or type(src)~="string" or src:sub(1,1)~="@" then return nil end
        local dir=src:match("^@(.*)[/\\][^/\\]+$")
        if dir and dir~="" then return dir end
        return nil
    end
    local function discoveryRoots()
        local roots={}
        local function add(p)
            if type(p)=="string" and p~="" then
                p=p:gsub("[\\/]+$","")
                if p~="" then
                    local lp=p:lower()
                    for _,e in ipairs(roots) do
                        if e:lower()==lp then return end
                    end
                    roots[#roots+1]=p
                end
            end
        end
        add(LOADER_FRAMES_DIR)
        add(scriptDir())
        if FileMgr and FileMgr.GetMenuRootPath then
            local ok,root=pcall(FileMgr.GetMenuRootPath)
            if ok and type(root)=="string" then
                add(root); add(root.."\\Lua")
                local parent=root:match("^(.*)[/\\][^/\\]+$")
                if parent then add(parent) end
            end
        end
        if type(os)=="table" and os.getenv then
            local okApp,app=pcall(os.getenv,"APPDATA")
            if okApp and type(app)=="string" then add(app.."\\Cherax"); add(app.."\\Cherax\\Lua") end
            local okProf,prof=pcall(os.getenv,"USERPROFILE")
            if okProf and type(prof)=="string" then add(prof.."\\Documents") end
        end
        return roots
    end
    local function probeBase(base)
        if not base or base=="" or not (FileMgr and FileMgr.DoesFileExist) then return nil end
        for _,chk in ipairs(PROBE_CHECKS) do
            local first=base..string.format(chk[1],FRAME_FIRST)
            local last=base..string.format(chk[1],FRAME_LAST)
            if FileMgr.DoesFileExist(first) and FileMgr.DoesFileExist(last) then
                return base..chk[2],chk[3]
            end
        end
        return nil
    end
    local function listFiles(dir,ext)
        if not dir or dir=="" or not (FileMgr and FileMgr.FindFiles) then return nil end
        local ok,res=pcall(FileMgr.FindFiles,dir,ext,true)
        if ok and type(res)=="table" and #res>0 then return res end
        ok,res=pcall(FileMgr.FindFiles,dir,"."..ext,true)
        if ok and type(res)=="table" and #res>0 then return res end
        ok,res=pcall(FileMgr.FindFiles,dir,"*."..ext,true)
        if ok and type(res)=="table" and #res>0 then return res end
        return nil
    end
    local function scanFor(marker,ext,fnamePattern)
        for _,root in ipairs(discoveryRoots()) do
            local files=listFiles(root,ext)
            if files then
                for _,f in ipairs(files) do
                    if type(f)=="string" and f:lower():match(marker) then
                        local dir=f:match("^(.*)[/\\][^/\\]+$")
                        if not dir and not f:find("[/\\]",1,true) then dir=root end
                        if dir and dir~="" then return dir,fnamePattern end
                    end
                end
            end
        end
        return nil
    end
    local function findFrames()
        for _,root in ipairs(discoveryRoots()) do
            local dir,pattern=probeBase(root)
            if dir then return dir,pattern end
        end
        local dir,pattern=scanFor("ezgif%-frame%-015%.png$","png","ezgif-frame-%03d.png")
        if dir then return dir,pattern end
        dir,pattern=scanFor("ezgif%-frame%-015%.jpg$","jpg","ezgif-frame-%03d.jpg")
        if dir then return dir,pattern end
        dir,pattern=scanFor("[/\\]015%.png$","png","%03d.png")
        if dir then return dir,pattern end
        return scanFor("[/\\]015%.jpg$","jpg","%03d.jpg")
    end
    local function startFrames()
        if framesState~="idle" then return end
        framesState="loading"
        if not (Texture and Texture.LoadTextureAsync and Texture.IsTextureValid and Texture.GetTexture) then
            framesState="failed"
            logLoader("Texture API unavailable; using logo fallback.")
            return
        end
        local ok,dir,pattern=pcall(findFrames)
        if not ok or not dir or not pattern then
            framesState="failed"
            logLoader("Frames "..FRAME_FIRST.."-"..FRAME_LAST.." not found; using logo fallback.")
            return
        end
        logLoader("Frames directory: "..dir)
        for i=FRAME_FIRST,FRAME_LAST do
            local path=dir.."\\"..string.format(pattern,i)
            local exists=FileMgr and FileMgr.DoesFileExist and FileMgr.DoesFileExist(path)
            if exists then
                local okId,id=pcall(Texture.LoadTextureAsync,path)
                if okId and type(id)=="number" then
                    frames[#frames+1]={id=id}
                else
                    framesMissing=framesMissing+1
                    logLoader(string.format("Failed to queue frame %03d.",i))
                end
            else
                framesMissing=framesMissing+1
                logLoader("Missing frame: "..string.format(pattern,i))
            end
        end
        if #frames==0 then
            framesState="failed"
            logLoader("No frames could be loaded; using logo fallback.")
            return
        end
        framesDeadline=now_ms()+10000
    end
    local function pruneFrames()
        local kept,keptN={},0
        for _,f in ipairs(frames) do
            if Texture.IsTextureValid(f.id) then keptN=keptN+1; kept[keptN]=f end
        end
        frames=kept
        return keptN
    end
    local function pollFrames()
        if framesState~="loading" then return end
        local valid,total=0,#frames
        for _,f in ipairs(frames) do
            if Texture.IsTextureValid(f.id) then valid=valid+1 end
        end
        if valid>=total then
            local keptN=pruneFrames()
            if keptN==0 then
                framesState="failed"
                logLoader("Frame textures invalid; using logo fallback.")
            else
                framesState="ready"
                resetFrameClock()
                if framesMissing==0 and keptN==FRAME_TOTAL then
                    logLoader(string.format("Loaded %d/%d frames successfully.",keptN,FRAME_TOTAL))
                else
                    logLoader(string.format("Loaded %d/%d frames (%d missing).",keptN,FRAME_TOTAL,FRAME_TOTAL-keptN))
                end
            end
        elseif now_ms()>framesDeadline then
            local keptN=pruneFrames()
            if keptN==0 then
                framesState="failed"
                logLoader("Frame load timeout (0 ready); using logo fallback.")
            else
                framesState="ready"
                resetFrameClock()
                logLoader(string.format("Loaded %d/%d frames (timeout).",keptN,FRAME_TOTAL))
            end
        end
    end
    local function advanceFrame(now)
        if not frameTick then frameTick=now return end
        local dt=now-frameTick
        frameTick=now
        if dt<=0 then return end
        if dt>500 then dt=500 end -- picos de tempo nao acumulam atraso de animacao
        frameAccum=frameAccum+dt
        local fps=type(LOADER_FPS)=="number" and LOADER_FPS>0 and LOADER_FPS or 24
        local guard=0
        while guard<60 do
            guard=guard+1
            local stepFps=fps
            if #frames>0 and frameIndex>#frames-LOADER_TAIL then stepFps=fps/LOADER_TAIL_DIV end
            local step=1000/math.max(1,stepFps)
            if frameAccum<step then break end
            frameAccum=frameAccum-step
            frameIndex=(frameIndex%#frames)+1
        end
    end
    -- advance=true anima (loop 015..050); advance=falso congela no ultimo frame.
    local function currentFrameSrv(now,advance)
        if framesState=="failed" or #frames==0 then return nil end
        if framesState=="loading" then pollFrames() end
        if framesState~="ready" then return nil end
        if advance then
            advanceFrame(now)
        else
            frameIndex=#frames
        end
        local f=frames[frameIndex] or frames[1]
        local tex=Texture.GetTexture(f.id)
        if not tex then return nil end
        local srv=tex:GetCurrent()
        if not srv then return nil end
        if not frameAspect then
            local okW,w=pcall(function() return tex:GetWidth() end)
            local okH,h=pcall(function() return tex:GetHeight() end)
            if okW and okH and type(w)=="number" and type(h)=="number" and w>0 and h>0 then
                frameAspect=w/h
            end
        end
        return srv
    end
    -- Encaixa a area de conteudo (FRAME_UV) dentro de um quadrado side x side, sem esticar.
    local function drawFrameContain(srv,x,y,side,alpha)
        local texAspect=(type(frameAspect)=="number" and frameAspect>0) and frameAspect or (16/9)
        local aspect=((FRAME_UV[3]-FRAME_UV[1])/(FRAME_UV[4]-FRAME_UV[2]))*texAspect
        local w,h=side,side/aspect
        if h>side then h=side; w=side*aspect end
        local fx,fy=x+(side-w)/2,y+(side-h)/2
        local color=(math.floor(math.max(0,math.min(1,alpha))*255)<<24)|0x00FFFFFF
        ImGui.AddImage(srv,fx,fy,fx+w,fy+h,FRAME_UV[1],FRAME_UV[2],FRAME_UV[3],FRAME_UV[4],color)
    end

    -- GitHub asset auto-install (logo + PHOTS frames) -------------------------
    -- Quando logo.png ou os frames PHOTS faltam no disco, baixa do repositorio,
    -- cria as pastas Lua\MeloCatScr e Lua\PHOTS e grava os arquivos.
    local ASSETS_REPO="https://raw.githubusercontent.com/melinskyRf/MeloMMrepo/main"
    local ASSETS_UA="MeloCatScr/1.0 (Cherax Lua)"
    local ASSETS_TIMEOUT=45000
    local PNG_MAGIC="\137PNG\r\n\26\n"
    local assetsState="idle" -- idle|busy|done|failed
    local assetsQueue,assetsRetry={},{}
    local assetsIdx,assetsCur,assetsCurAt=1,nil,0
    local assetsNeed,assetsFail,assetsPass=0,0,1
    local function assetsRoot()
        if not (FileMgr and FileMgr.GetMenuRootPath) then return nil end
        local ok,r=pcall(FileMgr.GetMenuRootPath)
        if ok and type(r)=="string" and r~="" then return r end
        return nil
    end
    local function assetsLogoPath()
        local root=assetsRoot()
        return root and (root.."\\Lua\\MeloCatScr\\logo.png") or nil
    end
    local function assetsPhotsDir()
        local root=assetsRoot()
        return root and (root.."\\Lua\\PHOTS") or nil
    end
    local function assetsExists(path)
        if not path or not (FileMgr and FileMgr.DoesFileExist) then return false end
        local ok,e=pcall(FileMgr.DoesFileExist,path)
        if not ok then return false end
        return not not e
    end
    local function assetsBuildQueue()
        local q={}
        local lp=assetsLogoPath()
        if lp and not assetsExists(lp) then q[#q+1]={url=ASSETS_REPO.."/logo.png",path=lp,name="logo.png"} end
        local pd=assetsPhotsDir()
        if pd then
            for i=FRAME_FIRST,FRAME_LAST do
                local fname=string.format("ezgif-frame-%03d.png",i)
                local p=pd.."\\"..fname
                if not assetsExists(p) then q[#q+1]={url=ASSETS_REPO.."/PHOTS/"..fname,path=p,name=fname} end
            end
        end
        return q
    end
    local function assetsStart()
        if assetsState=="busy" then return end
        if not (Curl and Curl.Easy and FileMgr and FileMgr.WriteFileContent and FileMgr.DoesFileExist) then
            assetsState="failed"
            logLoader("Asset install unavailable (Curl/FileMgr API missing).")
            return
        end
        assetsQueue=assetsBuildQueue()
        assetsRetry={}; assetsIdx=1; assetsCur=nil
        assetsFail=0; assetsPass=1; assetsNeed=#assetsQueue
        if assetsNeed==0 then assetsState="done"; return end
        if FileMgr.CreateDir then
            local root=assetsRoot()
            if root then
                pcall(FileMgr.CreateDir,root.."\\Lua\\MeloCatScr")
                pcall(FileMgr.CreateDir,root.."\\Lua\\PHOTS")
            end
        end
        assetsState="busy"
        logLoader(string.format("Missing %d asset(s); downloading from GitHub.",assetsNeed))
    end
    local function assetsFailItem(item,why)
        assetsFail=assetsFail+1
        assetsRetry[#assetsRetry+1]=item
        logLoader(string.format("Download failed: %s (%s).",item.name,why))
    end
    local assetsVerifyNote=false
    local function assetsSave(path,body)
        if io and io.open then
            local okf,f=pcall(io.open,path,"wb")
            if okf and f then
                local okw,w=pcall(function() return f:write(body) end)
                pcall(function() f:close() end)
                local written=okw and w~=nil
                local okr,g=pcall(io.open,path,"rb")
                if written and okr and g then
                    local okb,back=pcall(function() return g:read("*a") end)
                    pcall(function() g:close() end)
                    if okb and type(back)=="string" and #back==#body then return true end
                end
                if FileMgr and FileMgr.DeleteFile and assetsExists(path) then pcall(FileMgr.DeleteFile,path) end
            end
        end
        if not (FileMgr and FileMgr.WriteFileContent) then return false,"no writer" end
        local okw,rc=pcall(FileMgr.WriteFileContent,path,body,false)
        if not okw or rc~=true then return false,"write" end
        if FileMgr.DoesFileExist and not assetsExists(path) then return false,"missing" end
        if FileMgr.ReadFileContent then
            local okv,back=pcall(FileMgr.ReadFileContent,path)
            if okv and type(back)=="string" and #back==#body then return true end
            if not assetsVerifyNote then
                assetsVerifyNote=true
                local rb="error"
                if okv and type(back)=="string" then rb=tostring(#back) elseif okv then rb=tostring(back) end
                logLoader(string.format("Verify note: FileMgr read back %s of %d bytes; trusting WriteFileContent.",rb,#body))
            end
        end
        return true
    end
    local function assetsConsume()
        local item=assetsQueue[assetsIdx]
        local ok,finished=pcall(function() return assetsCur:GetFinished() end)
        if not ok then
            assetsFailItem(item,"curl state error")
        elseif not finished then
            assetsFailItem(item,"timeout")
        else
            local okr,code,body=pcall(function() return assetsCur:GetResponse() end)
            local curleOK=(eCurlCode and eCurlCode.CURLE_OK) or 6
            if not okr then
                assetsFailItem(item,"response error")
            elseif code~=curleOK then
                assetsFailItem(item,"code "..tostring(code))
            elseif type(body)~="string" or #body<128 or body:sub(1,8)~=PNG_MAGIC then
                assetsFailItem(item,"bad response")
            else
                local oksv,why=assetsSave(item.path,body)
                if not oksv then assetsFailItem(item,why or "save error") end
            end
        end
        assetsCur=nil
        assetsIdx=assetsIdx+1
    end
    local function assetsFinish()
        if assetsPass<2 and #assetsRetry>0 then
            assetsQueue=assetsRetry; assetsRetry={}; assetsIdx=1; assetsFail=0; assetsPass=2
            logLoader(string.format("Retrying %d failed download(s).",#assetsQueue))
            return
        end
        local haveLogo=assetsExists(assetsLogoPath())
        assetsState=haveLogo and "done" or "failed"
        logLoader(string.format("Assets ready: %d/%d (attempt %d).",assetsNeed-assetsFail,assetsNeed,assetsPass))
        if not haveLogo then logLoader("Logo unavailable; check the network and reload the script.") end
    end
    local function assetsPump()
        if assetsState~="busy" then return end
        if assetsCur then
            local item=assetsQueue[assetsIdx]
            local ok,finished=pcall(function() return assetsCur:GetFinished() end)
            local timedOut=(now_ms()-assetsCurAt)>ASSETS_TIMEOUT
            if ok and not finished and not timedOut then return end
            if ok and not finished then
                assetsFailItem(item,"timeout")
                assetsCur=nil
                assetsIdx=assetsIdx+1
            else
                assetsConsume()
            end
        end
        while assetsState=="busy" and assetsIdx<=#assetsQueue do
            local item=assetsQueue[assetsIdx]
            local launched=false
            local okc,c=pcall(Curl.Easy)
            if okc and c then
                local ok2=pcall(function()
                    if c.DisableErrorLog then c:DisableErrorLog() end
                    c:Setopt((eCurlOption and eCurlOption.CURLOPT_URL) or 4,item.url)
                    c:Setopt((eCurlOption and eCurlOption.CURLOPT_USERAGENT) or 5,ASSETS_UA)
                    c:Perform()
                end)
                if ok2 then
                    assetsCur=c; assetsCurAt=now_ms(); launched=true
                end
            end
            if launched then return end
            assetsFailItem(item,"setup error")
            assetsIdx=assetsIdx+1
        end
        if assetsState=="busy" and assetsIdx>#assetsQueue then assetsFinish() end
    end

        if start then
            local elapsed=math.max(0,now-start)
            local loaderTime,moveTime=3900,1150 -- 3900 = um ciclo completo com a cauda lenta
            if prefs.loader then
                local centerSide=math.min(math.max(finalSide*1.75,260),width*0.34,height*0.42)
                local centerX=(width-centerSide)/2
                local centerY=(height-centerSide)/2-36
                if elapsed<loaderTime then
                    local progress=math.max(0,math.min(1,elapsed/loaderTime))
                    local alpha=math.min(1,elapsed/450)
                    local srv=currentFrameSrv(now,true)
                    if srv then drawFrameContain(srv,centerX,centerY,centerSide,alpha) else image(centerX,centerY,centerSide,alpha) end
                    bar(centerY+centerSide+30,alpha,progress)
                    return
                elseif elapsed<loaderTime+moveTime then
                    local t=(elapsed-loaderTime)/moveTime; local ease=1-(1-t)^3
                    local side=centerSide+(finalSide-centerSide)*ease
                    local x=centerX+(finalX-centerX)*ease
                    local y=centerY+(finalY-centerY)*ease
                    image(x,y,side,1.0)
                    return
                else
                    played=true; preview=false; start=nil
                    if not prefs.fixed then return end
                end
            else
                local t=math.min(1,elapsed/650); local alpha=1-(1-t)^3
                if elapsed>4200 and not prefs.fixed then alpha=math.max(0,1-(elapsed-4200)/800) end
                if elapsed>=5000 then played=true; preview=false; start=nil; if not prefs.fixed then return end; alpha=1.0 end
                if elapsed<loaderTime then
                    local srv=currentFrameSrv(now,true)
                    if srv then drawFrameContain(srv,finalX,finalY,finalSide,alpha) else image(finalX,finalY,finalSide,alpha) end
                    bar(finalY+finalSide+30,alpha,math.max(0,math.min(1,elapsed/loaderTime)))
                else
                    image(finalX,finalY,finalSide,alpha)
                end
                return
            end
        end
        if prefs.fixed then image(finalX,finalY,finalSide,1.0) end
    end
    local id=EventMgr.RegisterHandler(eLuaEvent.ON_PRESENT,function()
        local ok,err=pcall(present)
        if not ok then failed=true; toast("Logo desativada nesta execucao: "..tostring(err)) end
    end)
    eventIds[#eventIds+1]=id
    -- The published Texture API has no unload method. Keep one managed texture per script,
    -- stop rendering on unload and let Cherax own GPU lifetime; never reload it every frame.
    util.on_stop(function() start=nil; preview=false; textureId=nil; frames={}; frameAspect=nil; frameIndex=1; frameAccum=0; frameTick=nil end)
    layoutRevision=layoutRevision+1
end
-- BRAND LOGO END

    -- Public bindings -------------------------------------------------------
    return {
        addBranding = { get = function() return addBranding end, set = function(value) addBranding = value end },
        uiGateOpen = { get = function() return uiGateOpen end, set = function(value) uiGateOpen = value end }
    }
end

end)()
