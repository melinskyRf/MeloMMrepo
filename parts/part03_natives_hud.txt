-- ----------------------------------------------------------------------------
-- [MC:presentation/navigation] 02.05 / NAVIGATION
-- Assemble navigation routes from registered feature sources.
-- ----------------------------------------------------------------------------
factories["presentation/navigation"]=(function()
-- Workspace assembly uses a route schema and immutable views of registered controls.
return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV=context:scope("presentation/navigation", {
        ["composeRoutes"]="presentation/view_composer",
        ["composeContacts"]="presentation/view_composer",
        ["routeSchema"]="presentation/route_schema",
        ["layoutRevision"]="presentation/workspace_state",
        ["menu"]="presentation/workspace_state",
        ["panel"]="presentation/workspace_state",
        ["playerRoots"]="presentation/workspace_state",
        ["root"]="presentation/workspace_state",
    })

    -- Implementation --------------------------------------------------------
    local function assembleWorkspace(parts)
        local index={}
        local function visit(node)
            index[node.path]=node
            for _,ref in ipairs(node.items) do if ref.child then visit(ref.child) end end
        end
        visit(root)
        local function source(path) return assert(index["MeloCatScr/"..path],"Missing registered section: "..path) end
        local home=menu.list(root,"Central",{},"Quick access to your workspace.")
        local function open(routeId)
            for i,ref in ipairs(root.navigation or {}) do
                if ref.child.routeId==routeId then panel.category=i-1; panel.filter=""; return end
            end
        end
        local function shortcut(key,label,description,routeId)
            local ref=menu.action(home,key,{},description,function() open(routeId) end)
            ref.name=label; ref.feature:SetName(label)
        end
    shortcut("Abrir bancada","Open Weapon Center","Arsenal, aiming, loadouts and ammunition.","weapons")
    shortcut("Explorar experiencias","Explore Fun Modes","Watch Dogs, Hitman and scene tools.","fun")
    shortcut("Personalizar visual","Personalize Interface","Colors, layout and workspace preferences.","settings")
        local sources={home=home,agent=parts.self,vehicle=parts.car,world=parts.world,powers=parts.hero,
            movement=source("Self/Movement"),travel=source("Self/Deslocamento"),gunsmith=source("Weapon Options/Gunsmith Loadouts"),
            combat=parts.combat,ammunition=parts.ammo,effects=parts.effects,weapons=parts.weapons,
            activities=parts.fun,watchDogs=source("Watch Dogs"),hitman=source("Fun Features/Hitman"),
            online=source("Online"),session=parts.lobby,defense=source("Protections"),settings=parts.settings}
        root.navigation=composeRoutes(routeSchema,sources)
        menu.redesignPlayer=composeContacts
        for _,node in pairs(playerRoots) do composeContacts(node) end
        panel.category=0; layoutRevision=layoutRevision+1
    end
    -- Public bindings -------------------------------------------------------
    return {assembleWorkspace={get=function() return assembleWorkspace end}}
end

end)()
-- [/MC:presentation/navigation]

-- [MC:presentation/preferences] 02.06 / PREFERENCES
-- Read, validate and apply interface preferences.
-- ----------------------------------------------------------------------------
factories["presentation/preferences"]=(function()
return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV = context:scope("presentation/preferences", {
        ["I18N"] = "presentation/localization",
        ["create"] = "presentation/control_registry",
        ["layoutRevision"] = "presentation/workspace_state",
        ["menu"] = "presentation/workspace_state",
        ["panel"] = "presentation/workspace_state",
        ["root"] = "presentation/workspace_state",

        ["uiRanges"] = "presentation/workspace_state",
        ["uiGroupRevision"] = "presentation/workspace_state",
        ["uiPrefs"] = "presentation/workspace_state"
    })


    -- Implementation --------------------------------------------------------
-- INTERFACE SETTINGS BEGIN
local function addInterfaceSettings(misc,funfeatures)
    -- Presentation moves keep original feature hashes, callbacks and Watch Dogs hotkeys.
    misc.name="Configuracoes"
    local links=menu.list(misc,"Links e creditos",{},"Links do MeloCatScr. Os botoes copiam os enderecos.")
    local keep={}
    for _,ref in ipairs(misc.items) do
        if ref.child==links then keep[#keep+1]=ref
        elseif ref.id then
            ref.parent=links; links.items[#links.items+1]=ref
            if ref.child then ref.child.parent=links; ref.child.name="Creditos" end
        end
    end
    misc.items=keep
    local visual=menu.list(misc,"Menu",{},"Visual e organizacao da interface MeloCatScr.")
    local playersPage=menu.list(misc,"Jogadores",{},"Aparencia da lista e do cabecalho de jogadores.")
    local preferences=menu.list(misc,"Preferencias",{},"Salva apenas ajustes visuais; nao salva ativacoes de gameplay.")
    misc.viewTabs={{name="Menu",node=visual},{name="Jogadores",node=playersPage},{name="Preferencias",node=preferences},{name="Links e creditos",node=links}}
    local wdRef
    for i,ref in ipairs(root.items) do if ref.child and ref.child.name=="Watch Dogs" then wdRef=ref; table.remove(root.items,i); break end end
    if wdRef then
        wdRef.parent=funfeatures; wdRef.child.parent=funfeatures; wdRef.viewOnly=true
        funfeatures.items[#funfeatures.items+1]=wdRef
        funfeatures.viewTabs={{name="Geral",node=funfeatures},{name="Watch Dogs",node=wdRef.child}}
    end
    layoutRevision=layoutRevision+1
    local controls={}
    local keys={"color","hints","spacing","search","groups","swap","playerWidth","playerNames","playerIds",
        "uiScale","sidebarW","iconSize","cardH","btnH","gapY","radius","borderS","glowS","shadowS",
        "panelOpacity","panelContrast","hoverBoost","accentPower","animSpeed","scrollbarW"}
    local function changed(key,value)
        uiPrefs[key]=value
        if key=="groups" then uiGroupRevision=uiGroupRevision+1 end
        if key=="search" and value==0 then panel.filter="" end
    end
    local function toggle(group,key,name,description)
        local r=menu.toggle(group,name,{},description,function(value) changed(key,value and 1 or 0) end)
        r.value=uiPrefs[key]==1; r.feature:SetBoolValue(r.value); controls[key]=r
    end
    local function slider(group,key,name,description)
        local range=uiRanges[key] or {0,100}
        local r=create(group,name,eFeatureType.SliderInt,description,function(f)
            changed(key,math.max(range[1],math.min(range[2],f:GetIntValue())))
        end)
        r.feature:SetLimitValues(range[1],range[2]):SetStepSize(range[2]-range[1]>40 and 5 or 1):SetIntValue(uiPrefs[key])
        controls[key]=r
        return r
    end
    local function applyValues(values)
        for _,key in ipairs(keys) do
            if values[key]~=nil then
                changed(key,values[key]); local r=controls[key]
                if r then
                    r.generation=r.generation+1
                    if r.kind==eFeatureType.Toggle then r.value=values[key]==1; r.feature:SetBoolValue(r.value)
                    elseif key=="color" then r.feature:SetListIndex(values[key]-1)
                    else r.feature:SetIntValue(values[key]) end
                end
            end
        end
    end
    -- Settings organization: Visual, Layout, Typography and Advanced categories.
    local groupVisual=menu.list(visual,"Visual",{},"Cores, bordas, brilho e profundidade da interface.")
    local groupLayout=menu.list(visual,"Layout",{},"Tamanhos, espacamento e presets da interface.")
    local groupType=menu.list(visual,"Tipografia",{},"Escala geral da interface.")
    local groupAdv=menu.list(visual,"Avancado",{},"Animacoes e acabamento avancado.")
    local colorRange=uiRanges.color or {1,3}
    local theme=create(groupVisual,"Cor dos titulos",eFeatureType.Combo,"Cor de destaque dos titulos, contornos e controles do MeloCatScr.",function(f)
        changed("color",math.max(colorRange[1],math.min(colorRange[2],f:GetListIndex()+1)))
    end)
    theme.feature:SetList({"Roxo MeloCat","Magenta neon","Roxo profundo","Azul eletrico","Dourado","Verde melancia"}):SetListIndex(uiPrefs.color-1); controls.color=theme
    toggle(groupVisual,"hints","Mostrar dicas","Exibe explicacoes curtas acima das colunas.")
    slider(groupVisual,"accentPower","Intensidade do roxo","Forca da cor de destaque em titulos, contornos, barra ativa e botoes.")
    slider(groupVisual,"panelOpacity","Transparencia dos paineis","Opacidade dos fundos internos; valores menores deixam o preto do fundo aparecer.")
    slider(groupVisual,"panelContrast","Contraste dos paineis","Luminosidade dos fundos internos; aumente para separar melhor os blocos.")
    slider(groupVisual,"borderS","Intensidade da borda","Forca do contorno fino ao redor de caixas e grupos.")
    slider(groupVisual,"radius","Arredondamento","Raio dos cantos das caixas, campos e botoes.")
    slider(groupVisual,"glowS","Glow das bordas","Brilho do destaque da barra ativa, bordas de grupos e foco de busca.")
    slider(groupVisual,"shadowS","Sombra","Sombra suave atras do cabecalho, barra lateral e area de conteudo.")
    slider(groupVisual,"hoverBoost","Elevacao do hover","Quanto os itens se destacam mais ao passar o mouse.")
    toggle(groupLayout,"spacing","Espaco entre grupos","Adiciona respiro entre os grupos. Desligue para uma lista mais compacta.")
    toggle(groupLayout,"search","Mostrar busca","Busca por nome da opcao ou grupo na pagina aberta. Desligar limpa a busca.")
    toggle(groupLayout,"swap","Inverter ordem dos grupos","Troca a posicao de acoes e ajustes dentro de cada grupo. Padrao: ajustes primeiro.")
    toggle(groupLayout,"groups","Grupos abertos","Abre ou recolhe todos os grupos; depois voce pode expandir cada um individualmente.")
    menu.action(groupLayout,"Limpar buscas",{},"Limpa a busca de opcoes e o filtro de nomes da lista de jogadores.",function() panel.filter=""; panel.search="" end)
    slider(groupLayout,"sidebarW","Largura da sidebar","Lateral expandida em pixels; a versao recolhida mantem o tamanho compacto.")
    slider(groupLayout,"cardH","Altura dos cards","Escala da altura dos cards de armas, cenas e status.")
    slider(groupLayout,"btnH","Altura dos botoes","Altura dos botoes de destaque e do cabecalho.")
    slider(groupLayout,"gapY","Espacamento entre itens","Respiro vertical entre controles e grupos.")
    slider(groupLayout,"iconSize","Tamanho dos icones","Escala dos icones da barra lateral.")
    slider(groupLayout,"scrollbarW","Largura da scrollbar","Espessura da barra de rolagem.")
    slider(groupType,"uiScale","Escala da UI","Escala geral dos textos e controles; em janelas pequenas pode cortar conteudo.")
    slider(groupAdv,"animSpeed","Velocidade das animacoes","Ritmo das transicoes de hover, barra ativa e troca de pagina.")
    local presets={
        {"Compacto",{uiScale=92,sidebarW=132,cardH=94,btnH=24,gapY=5,radius=8,scrollbarW=7,iconSize=13}},
        {"Padrao",{uiScale=100,sidebarW=165,cardH=100,btnH=28,gapY=9,radius=10,scrollbarW=9,iconSize=14,
            borderS=65,glowS=40,shadowS=35,panelOpacity=98,panelContrast=100,hoverBoost=50,animSpeed=100,accentPower=100}},
        {"Confortavel",{uiScale=105,sidebarW=180,cardH=105,btnH=30,gapY=11,radius=11,scrollbarW=10,iconSize=15}},
        {"Grande",{uiScale=118,sidebarW=210,cardH=112,btnH=33,gapY=13,radius=12,scrollbarW=11,iconSize=17}},
        {"Premium",{uiScale=104,sidebarW=176,cardH=104,btnH=30,gapY=10,radius=12,scrollbarW=9,iconSize=15,
            borderS=80,glowS=70,shadowS=55,panelOpacity=96,panelContrast=106,hoverBoost=75,animSpeed=120,accentPower=115}},
    }
    for _,preset in ipairs(presets) do
        local name,values=preset[1],preset[2]
        menu.action(groupLayout,"Preset "..name,{},"Aplica o conjunto de ajustes "..name..". Use Salvar visual para guardar.",function()
            applyValues(values)
            toast(I18N.text("Preset aplicado. Use Salvar visual para guardar."))
        end)
    end
    local width=create(playersPage,"Largura da lista",eFeatureType.SliderInt,"Largura desejada em pixels; limitada a 45% do espaco disponivel.",function(f)
        changed("playerWidth",math.max(110,math.min(240,f:GetIntValue())))
    end)
    width.feature:SetLimitValues(110,240):SetStepSize(5):SetIntValue(uiPrefs.playerWidth); controls.playerWidth=width
    toggle(playersPage,"playerNames","Mostrar nomes","Aplica apenas a lista e o cabecalho do MeloCatScr. Desligado usa Jogador + numero; outros paineis e toasts tem ajustes proprios.")
    toggle(playersPage,"playerIds","Mostrar IDs","Mostra o numero do slot ao lado do nome.")
    local function decode(text)
        if type(text)~="string" or #text>4096 then return nil end
        text=text:gsub("\r","")
        local values={};local version
        for line in text:gmatch("[^\n]+") do
            local key,value=line:match("^([%w_]+)=(%d+)$")
            if not key or values[key]~=nil or (key~="version" and uiDefaults[key]==nil) then return nil end
            value=tonumber(value)
            if key=="version" then
                if value~=1 and value~=2 then return nil end
                version=value
            else
                local range=uiRanges[key]
                if range then if value<range[1] or value>range[2] then return nil end
                elseif value~=0 and value~=1 then return nil end
            end
            values[key]=value
        end
        if not version then return nil end
        -- Version 1 files predate the visual keys; they keep the original nine and fall back to defaults.
        local required=version==2 and keys
            or {"color","hints","spacing","search","groups","swap","playerWidth","playerNames","playerIds"}
        for _,key in ipairs(required) do if values[key]==nil then return nil end end
        for _,key in ipairs(keys) do if values[key]==nil then values[key]=uiDefaults[key] end end
        return values
    end
    local function path()
        return meloDataPath("MeloCatScrInterface.ini")
    end
    local function loadPrefs(quiet)
        local file=meloReadPath("MeloCatScrInterface.ini")
        if not file or not FileMgr.ReadFileContent then if not quiet then toast(I18N.text("Leitura de preferencias indisponivel.")) end; return end
        if FileMgr.DoesFileExist and not FileMgr.DoesFileExist(file) then return end
        local ok,text=pcall(FileMgr.ReadFileContent,file)
        local values=ok and decode(text)
        if values then applyValues(values); if not quiet then toast(I18N.text("Preferencias carregadas.")) end
        elseif not quiet then toast(I18N.text("Arquivo de preferencias ausente ou invalido. Ajustes atuais preservados.")) end
    end
    menu.action(preferences,"Salvar visual",{},"Grava MeloCatScrInterface.ini e carrega esses ajustes ao iniciar. Nao grava poderes, VPN, alvos ou nomes.",function()
        local file=path(); if not file or not FileMgr.WriteFileContent then toast(I18N.text("Gravacao de preferencias indisponivel.")); return end
        local lines={"version=2"}; for _,key in ipairs(keys) do lines[#lines+1]=key.."="..uiPrefs[key] end
        local ok,result=pcall(FileMgr.WriteFileContent,file,table.concat(lines,"\n").."\n",false)
        if ok and result==true then meloDropLegacy("MeloCatScrInterface.ini") end
        toast(ok and result==true and "Visual salvo em MeloCatScrInterface.ini." or "Nao foi possivel salvar o visual.")
    end)
    menu.action(preferences,"Carregar visual",{},"Rele os ajustes visuais salvos. Arquivos invalidos nao sao executados nem aplicados.",function() loadPrefs(false) end)
    menu.action(preferences,"Restaurar visual",{},"Volta aos padroes da interface e limpa as buscas. Clique Salvar visual se quiser guardar os padroes.",function()
        applyValues(uiDefaults); panel.filter=""; panel.search=""; toast(I18N.text("Visual restaurado; funcoes do jogo preservadas."))
    end)
    loadPrefs(true)
end
-- INTERFACE SETTINGS END

    -- Public bindings -------------------------------------------------------
    return {
        addInterfaceSettings = { get = function() return addInterfaceSettings end, set = function(value) addInterfaceSettings = value end }
    }
end

end)()

-- ----------------------------------------------------------------------------
-- [MC:presentation/view_composer] 02.09 / VIEW COMPOSER
-- Project control trees into independent navigation views.
-- ----------------------------------------------------------------------------
factories["presentation/view_composer"]=(function()
-- Render trees are projections of control trees, not mutated registration trees.
return function(context)
    -- Dependencies ----------------------------------------------------------
    local _ENV=context:scope("presentation/view_composer", {
        ["sectionLabels"]="presentation/route_schema",
        ["contactRoutes"]="presentation/route_schema",
    })

    -- Implementation --------------------------------------------------------
    local function project(source,excluded,trail)
        if excluded[source] then return nil end
        assert(not trail[source],"Navigation cycle: "..source.path)
        trail[source]=true
        local result={name=sectionLabels[source.name] or source.name,path=source.path,items={},pid=source.pid,source=source}
        for _,ref in ipairs(source.items) do
            if ref.child then
                local child=project(ref.child,excluded,trail)
                if child then result.items[#result.items+1]={name=child.name,child=child} end
            else result.items[#result.items+1]=ref end
        end
        trail[source]=nil
        return result
    end
    local function composeContacts(node)
        local indexed,ownership={},{}
        local function index(current)
            indexed[current.path:sub(#node.path+2)]=current
            for _,ref in ipairs(current.items) do if ref.child then index(ref.child) end end
        end
        index(node)
        for _,spec in ipairs(contactRoutes) do
            for _,path in ipairs(spec.sources) do
                local source=indexed[path]
                if source then ownership[source]=spec.id end
            end
        end
        local items={}
        for _,spec in ipairs(contactRoutes) do
            local page={name=spec.name,path=node.path.."/Views/"..spec.id,items={},pid=node.pid}
            local excluded={}
            for source,owner in pairs(ownership) do if owner~=spec.id then excluded[source]=true end end
            for _,path in ipairs(spec.sources) do
                local source=indexed[path]
                if source then
                    local child=project(source,excluded,{})
                    if child then page.items[#page.items+1]={name=child.name,child=child} end
                end
            end
            items[#items+1]={name=page.name,child=page}
        end
        -- Keep future unassigned categories reachable without duplicating known pages.
        local remaining={}
        for source in pairs(ownership) do remaining[source]=true end
        local extra=project(node,remaining,{})
        if #extra.items>0 then
            extra.name="Additional Tools";items[#items+1]={name=extra.name,child=extra}
        end
        node.navigation=items
    end
    local function composeRoutes(schema,sources)
        local items={}
        for _,spec in ipairs(schema) do
            local route={name=spec.name,path="MeloCatScr/Workspace/"..spec.id,routeId=spec.id,
                subtitle=spec.hint,items={},isPlayersCategory=spec.players}
            if not spec.players then
                route.viewTabs={}
                for _,definition in ipairs(spec.pages) do
                    local excluded={}
                    for _,key in ipairs(definition.exclude or {}) do if sources[key] then excluded[sources[key]]=true end end
                    local page={name=definition.name,path=route.path.."/"..definition.id,items={}}
                    for _,key in ipairs(definition.sources) do
                        local source=assert(sources[key],"Navigation source missing: "..key)
                        local tree=project(source,excluded,{})
                        if tree then page.items[#page.items+1]={name=tree.name,child=tree} end
                    end
                    route.viewTabs[#route.viewTabs+1]={name=page.name,node=page}
                end
            end
            items[#items+1]={name=route.name,child=route}
        end
        return items
    end
    -- Public bindings -------------------------------------------------------
    return {composeRoutes={get=function() return composeRoutes end},composeContacts={get=function() return composeContacts end}}
end

end)()

local function statsFor(key, family)
    local base = familyProfiles[family] or familyProfiles.Experimental
    local seed = 7
    key = tostring(key or family or "")
    for i = 1, #key do seed = (seed*31 + key:byte(i)) % 99991 end
    local out = {}
    for i, value in ipairs(base) do
        local jitter = math.floor(seed / (2^i)) % 11
        out[i] = math.max(8, math.min(99, value - 5 + jitter))
    end
    return out
end

local function gatherControls(node, into)
    for _, ref in ipairs(node.items) do
        if ref.child then gatherControls(ref.child, into)
        elseif ref.id then into[#into + 1] = ref end
    end
end
local function isInput(ref)
    return ref.kind and ref.kind ~= eFeatureType.Button and ref.kind ~= eFeatureType.Toggle
end
local function buildLayout(node)
    if node.layout and node.layout.revision == layoutRevision then return node.layout end
    local layout = {quick = {}, configured = {}, revision = layoutRevision}
    local included,allControls={},{}
    gatherControls(node,allControls)
    for _,ref in ipairs(allControls) do included[ref]=true end
    local function add(side, title, controls)
        -- An alias must not draw the same ImGui ID twice when its original is on this page.
        local unique={}
        for _,ref in ipairs(controls) do
            if not (ref.aliasOf and included[ref.aliasOf]) then unique[#unique+1]=ref end
        end
        controls=unique
        if #controls == 0 then return end
        local inputs = 0
        for _, ref in ipairs(controls) do if isInput(ref) then inputs = inputs + 1 end end
        local groups = layout[side]
        groups[#groups + 1] = {title = title, controls = controls, inputs = inputs, order = #groups + 1}
    end
    local function collect(current, prefix)
        local paired, settings, directInputs, configToggle = {}, {}, 0, false
        for _, ref in ipairs(current.items) do
            if ref.child then
                local sourceName=ref.child.source and ref.child.source.name or ref.name
                local featureName = sourceName:match("^(.-) Settings$")
                if featureName then
                    for _, sibling in ipairs(current.items) do
                        if sibling.id and not sibling.child and sibling.name == featureName then
                            settings[ref] = sibling; paired[sibling] = true; break
                        end
                    end
                end
            elseif isInput(ref) then
                directInputs = directInputs + 1
                -- Independent selectors configure a system; paired sliders configure only their action.
                if not ref.configGroup then configToggle = true end
            end
        end
        local quick, configured, actionGroups, groupOrder = {}, {}, {}, {}
        local system = configToggle or directInputs > 1 or (current.source and current.source.name or current.name):match(" Settings$") ~= nil
        for _, ref in ipairs(current.items) do
            if ref.child then
                local title = prefix == "" and ref.name or prefix .. " / " .. ref.name
                if settings[ref] then
                    local controls = {settings[ref]}
                    gatherControls(ref.child, controls)
                    add("configured", title:gsub(" Settings$", ""), controls)
                elseif not ref.viewOnly then collect(ref.child, title) end
            elseif ref.id and not paired[ref] then
                local needsConfig = isInput(ref) or ref.configGroup or configuredActions[ref.name]
                    or (system and ref.kind == eFeatureType.Toggle)
                if needsConfig then
                    if ref.configGroup and not system then
                        local key = ref.configGroup
                        if not actionGroups[key] then actionGroups[key] = {}; groupOrder[#groupOrder + 1] = key end
                        actionGroups[key][#actionGroups[key] + 1] = ref
                    else configured[#configured + 1] = ref end
                else quick[#quick + 1] = ref end
            end
        end
        local title = prefix ~= "" and prefix or current.name
        add("quick", title, quick)
        add("configured", title, configured)
        for _, group in ipairs(groupOrder) do add("configured", title .. " / " .. group.name, actionGroups[group]) end
    end
    collect(node, "")
    table.sort(layout.configured, function(a, b)
        if a.inputs ~= b.inputs then return a.inputs > b.inputs end
        return a.order < b.order
    end)
    node.layout = layout
    return layout
end
local function renderControl(ref)
    if ref.renderFailure and ref.renderFailure.id~=ref.id then ref.renderFailure=nil end
    if not ref.renderFailure then
        local ok,err=pcall(function()
            -- Never reuse a cached native object if the core feature disappeared or changed type.
            if ref.cheraxNative and FeatureMgr.GetFeature then
                local current=FeatureMgr.GetFeature(ref.id)
                assert(current and current:GetHash()==ref.id and current:GetType()==ref.kind
                    and not current:IsPlayerFeature(),"Cherax control changed or is no longer available")
            end
            ClickGUI.RenderFeature(ref.id)
        end)
        if not ok then
            ref.renderFailure={id=ref.id,message=tostring(err)}
            -- Only once per failed control; no native or toast error flood every frame.
            pcall(report,"Interface / "..ref.path..": "..tostring(err))
        end
    end
    if ref.renderFailure then
        ImGui.TextWrapped("%s",I18N.text(ref.name).." - "..I18N.text("Interface error. Details: MeloCatScrErrors.txt"))
        if ImGui.Button(I18N.text("Retry control").."##MeloCatScrRetry"..tostring(ref.id)) then
            ref.renderFailure=nil
        end
    end
end
-- Personal library data never executes code and never saves gameplay activation.
local function libraryPath()
    return meloDataPath("MeloCatScrLibrary.ini")
end
local function loadLibrary()
    local path=meloReadPath("MeloCatScrLibrary.ini")
    if not path or not FileMgr.ReadFileContent then return end
    if FileMgr.DoesFileExist and not FileMgr.DoesFileExist(path) then return end
    local ok,text=pcall(FileMgr.ReadFileContent,path)
    if not ok or type(text)~="string" or #text>65536 or not text:match("^version=1[\r\n]")then return end
    panel.weaponFavorites=panel.weaponFavorites or {}
    for line in text:gmatch("[^\r\n]+")do
        local id=line:match("^favorite=(%d+)$")
        local hash=line:match("^weapon=(WEAPON_[%w_]+)$")
        if id then panel.favorites[tonumber(id)]=true end
        if hash then panel.weaponFavorites[hash]=true end
    end
end
local function saveLibrary()
    local path=libraryPath()
    if not path or not FileMgr.WriteFileContent then return end
    local lines={"version=1"}
    for id,enabled in pairs(panel.favorites)do if enabled then lines[#lines+1]="favorite="..id end end
    for hash,enabled in pairs(panel.weaponFavorites or {})do if enabled then lines[#lines+1]="weapon="..hash end end
    local text=table.concat(lines,"\n").."\n"
    context:read("application/task_dispatcher","enqueue")(function()
        local ok,result=pcall(FileMgr.WriteFileContent,path,text,false)
        if ok and result==true then meloDropLegacy("MeloCatScrLibrary.ini")
        else report("Personal library could not be saved; favorites remain available in this session.")end
    end)
end
loadLibrary()

    if ImGui.Button((active and "Saved" or "+ Favorite").."##favorite"..ref.id) then
        panel.favorites[ref.id]=not active or nil;saveLibrary()
    end
end

local function renderControlGroups(node,id)
    local layout=buildLayout(node)
    local query=(panel.filter or ""):lower()
    local shown=0
    local order=uiPrefs.swap==1 and {layout.configured,layout.quick} or {layout.quick,layout.configured}
    for _,groups in ipairs(order) do
        for index,group in ipairs(groups) do
            local matches={}
            for _,ref in ipairs(group.controls) do
                if query=="" or I18N.text(group.title):lower():find(query,1,true) or I18N.text(ref.name):lower():find(query,1,true) then matches[#matches+1]=ref end
            end
            if #matches>0 then
                if uiPrefs.spacing~=0 and shown>0 then ImGui.Spacing() end
                shown=shown+#matches
                local groupFlags=uiPrefs.groups==1 and (ImGuiTreeNodeFlags and ImGuiTreeNodeFlags.DefaultOpen or 32) or 0
                local open=ImGui.CollapsingHeader(group.title.."  /  "..#matches.."##"..id..group.order..tostring(groups==layout.quick).."-"..uiGroupRevision,groupFlags)
                if open or query~="" then
                    local npush=0
                    if ImGui.PushStyleColor and ImGuiCol and ImGuiCol.Border then
                        local c=uiAccent()
                        ImGui.PushStyleColor(ImGuiCol.Border,c[1]*0.40,c[2]*0.36,c[3]*0.52,uiGlowAlpha(0.85))
                        npush=npush+1
                    end
                    npush=npush+pushCardBg()
                    local childH=uiCardHeight(math.min(360,44+#matches*72))
                    local visible=ImGui.BeginChild("ControlGroup##"..id..group.order..tostring(groups==layout.quick),0,childH,true,0)
                    if visible then
                        uiFontScale()
                        for _,ref in ipairs(matches) do
                            ImGui.PushItemWidth(-1);renderControl(ref);ImGui.PopItemWidth()
                            favorite(ref);ImGui.Spacing()
                        end
                    end
                    ImGui.EndChild()
                    if npush>0 then ImGui.PopStyleColor(npush) end
                end
            end
        end
    end
    if shown==0 then ImGui.TextDisabled("No matching controls.") end
end

local function renderLibrary(mode)
    local query=(panel.globalSearch or ""):lower();local list={}
    if mode=="recent" then
        local refs=context:read("presentation/workspace_state","refs")
        for _,id in ipairs(panel.recent) do local r=refs[id];if r and not panel.favorites[r.id] and (not r.pid or panel.active[r.pid]) then list[#list+1]=r end end
    else
        for _,ref in ipairs(allControls()) do
            if (not ref.pid or panel.active[ref.pid]) and (mode~="favorites" or panel.favorites[ref.id]) and (query=="" or I18N.text(ref.name):lower():find(query,1,true) or ref.path:lower():find(query,1,true)) then list[#list+1]=ref end
        end
    end
    if mode=="search" and query~="" then
        local sug={}
        for _,ref in ipairs(list) do
            local name=I18N.text(ref.name)
            if name:lower()~=query and #sug<6 then sug[#sug+1]={label=name} end
        end
        suggestionStrip(sug,function(item) panel.globalSearch=item.label end,"global")
    end
    local pageKey="results:"..mode..":"..query
    panel.resultPage=panel.resultKey==pageKey and (panel.resultPage or 1) or 1;panel.resultKey=pageKey
    local pages=math.max(1,math.ceil(#list/12));panel.resultPage=math.min(panel.resultPage,pages)
    ImGui.TextDisabled(#list.." controls / page "..panel.resultPage.." of "..pages)
    if panel.resultPage>1 and ImGui.Button("Previous##results") then panel.resultPage=panel.resultPage-1 end
    if panel.resultPage<pages and ImGui.Button("Next##results") then panel.resultPage=panel.resultPage+1 end
    for i=(panel.resultPage-1)*12+1,math.min(#list,panel.resultPage*12) do featureCard(list[i],mode) end
    if #list==0 then
        if mode=="search" then ImGui.TextWrapped("No controls match this search.")
        else ImGui.TextWrapped("Mark a control as Favorite or use a control to see it here.") end
    end
end

local function homeIpText()
    local ok,addr=pcall(ipStringFor,panel.localPlayer)

end

local function renderContentBody(route,stateKey,id)
    local pageIndex=panel["view:"..route.path] or 0
    if route.viewTabs then
        pageIndex=math.max(0,math.min(pageIndex,#route.viewTabs-1))
        panel["view:"..route.path]=pageIndex
    end
    local page=route.viewTabs and route.viewTabs[pageIndex+1]
    ImGui.TextDisabled("MELOCAT / "..string.upper(route.name)..(page and (" / "..string.upper(page.name)) or ""))
    uiTitle(page and page.name or route.name)
    if uiPrefs.hints~=0 and route.subtitle then ImGui.TextDisabled(route.subtitle) end
    if panel.globalSearch~="" then renderLibrary("search");return end
    if panel.library then renderLibrary(panel.library);return end
    if route.isPlayersCategory then renderPlayersPanel(id);return end
    if route.viewTabs and #route.viewTabs>1 then
        local names={}
        for _,entry in ipairs(route.viewTabs)do names[#names+1]=entry.name end
        local chosen=filterChips(names,names[pageIndex+1],"pages"..id)
        for i,name in ipairs(names)do
            if name==chosen and panel["view:"..route.path]~=i-1 then
                panel["view:"..route.path]=i-1;pageIndex=i-1;page=route.viewTabs[i]
            end
        end
        ImGui.Spacing()
    end
    if not page then return end
    -- Weapon pages own their search; every other page gets the shared filter.
    if route.routeId~="weapons" then pageFilterInput(page.node) end
    if route.routeId=="home" then renderHome(page.node);return end
    if route.routeId=="weapons" then
        if page.name=="Weapon Center" then renderWeaponCenter();return end
        if page.name=="Loadouts" then renderLoadouts();return end
        if page.name=="Special Gear" then renderSpecial(page.node,id);return end
    end
    renderControlGroups(page.node,id.."/"..page.node.path)
end

    local target=collapsed and 60 or math.max(120,math.min(220,uiPrefs.sidebarW or 165))
    panel.sidebarW=panel.sidebarW or target

    local function pushBg(r,g,b)
        if ImGui.PushStyleColor and ImGuiCol and ImGuiCol.ChildBg then
            ImGui.PushStyleColor(ImGuiCol.ChildBg,math.min(1,r*contrast),math.min(1,g*contrast),math.min(1,b*contrast),opacity)
            return 1
        end
        return 0
    end

        if ImGui.PushStyleColor and ImGuiCol and ImGuiCol.Border then
            ImGui.PushStyleColor(ImGuiCol.Border,color[1],color[2],color[3],alpha)
            return 1
        end
        return 0
    end
    local function drawShadow(w,h)
        if not (iconsOk and ImGui.AddRectFilled and ImGui.GetCursorScreenPos and h) then return end
        local alpha=uiShadowAlpha()
        if alpha<=0.01 or h<=0 then return end
